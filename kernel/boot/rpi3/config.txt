# config.txt -- goes on the SD card's FAT32 boot partition, next to
# kernel8.img and the firmware (bootcode.bin, start.elf, fixup.dat).
# This is what replaces GRUB + the Multiboot header on x86.
#
# Board: Raspberry Pi 3 Model B+

# Boot the 64-bit kernel8.img at 0x80000 (boot/rpi3/linker.ld)
arm_64bit=1
kernel=kernel8.img
kernel_address=0x80000

# PL011 UART0 on GPIO 14/15 instead of Bluetooth, 48 MHz UART clock
# (uart.c computes the baud divisor from it): 115200 8N1
enable_uart=1
dtoverlay=disable-bt
init_uart_clock=48000000

# The framebuffer will be requested over the mailbox; these are its defaults
disable_overscan=1
framebuffer_depth=32
